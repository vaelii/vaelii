;; CxSocialExtension — an opt-in extension of CxSocial: relationship vocabulary
;; narrower than CxSocial's own. Sees CxSocial, which carries it to the shared
;; upper ontology via CxUniverse. CxWell does not see it: a context opts in by
;; placing itself under CxSocialExtension.
;;
;; "<Theory>Extension" names a theory that extends an existing starter theory
;; with vocabulary most contexts under the base theory have no occasion to see:
;; the base theory ships to everyone below CxWell, the extension only to a
;; context placed under it.

(genlCx CxSocialExtension CxSocial)

;; ---- nestingPartnerOf -------------------------------------------------------
(comment nestingPartnerOf
         "(nestingPartnerOf ?x ?y) means ?x and ?y are romantic partners who live together. A spec of romanticPartnerOf and of roommateOf, and read by the rule below from the two together.")
(set/monotonic (instance_relation_predicate nestingPartnerOf))
(set/monotonic (symmetric nestingPartnerOf))
(arg nestingPartnerOf 1 person)
(arg nestingPartnerOf 2 person)
(genl nestingPartnerOf romanticPartnerOf)
(genl nestingPartnerOf roommateOf)

(set/forwardRule
  (implies (and (romanticPartnerOf ?x ?y) (roommateOf ?x ?y))
           (nestingPartnerOf ?x ?y)))

;; ---- chosen kinship ----------------------------------------------------------
(comment chosenSiblingOf
         "(chosenSiblingOf ?x ?y) means ?x and ?y are siblings by choice, not by birth: family by choice, so a spec of relativeOf. Neither implies nor is implied by the biological siblingOf.")
(set/monotonic (instance_relation_predicate chosenSiblingOf))
(set/monotonic (time_varying_relation chosenSiblingOf))
(set/monotonic (symmetric chosenSiblingOf))
(arg chosenSiblingOf 1 person)
(arg chosenSiblingOf 2 person)
(genl chosenSiblingOf relativeOf)
