;; CxKinship — kinship & age theory: the genl edge that derives ancestorOf from parentOf,
;; the genlInverse edges that answer parentOf from fatherOf and motherOf, and the rules
;; that derive grandparentOf from parentOf and olderThan from birthYearOf. Sees the shared
;; upper ontology via CxUniverse; CxWell sees it.

(genlCx CxKinship CxUniverse)
(genlCx CxWell CxKinship)

;; Parenthood is ancestry: parentOf specializes ancestorOf.
(genl parentOf ancestorOf)

(set/forwardRule (implies (and (parentOf ?x ?y) (parentOf ?y ?z)) (grandparentOf ?x ?z)))

(set/backwardRule
  (implies (and (birthYearOf ?x ?bx) (birthYearOf ?y ?by) (lessThan ?bx ?by))
           (olderThan ?x ?y)))

;; Maternity and paternity each entail parenthood with the arguments exchanged, and
;; neither converse holds: a parent is one or the other, and nothing in a parentOf fact
;; says which.
(genlInverse fatherOf parentOf)
(genlInverse motherOf parentOf)

;; The same carrier can carry unrelated children in separate pregnancies. Sharing
;; that parent alone proves neither siblinghood nor its blood-qualified reading.
(exceptWhen (gestationalOnlyParentOf ?parent ?first)
 (set/defaultRule
  (set/forwardRule
   (implies (and (parentOf ?parent ?first) (parentOf ?parent ?second)
                 (different ?first ?second))
            (siblingOf ?first ?second)))))
(exceptWhen (gestationalOnlyParentOf ?parent ?second)
 (set/defaultRule
  (set/forwardRule
   (implies (and (parentOf ?parent ?first) (parentOf ?parent ?second)
                 (different ?first ?second))
            (siblingOf ?first ?second)))))

(exceptWhen (gestationalOnlyParentOf ?parent ?first)
 (set/defaultRule
  (set/forwardRule
   (implies (and ((BloodKinFn parentOf) ?parent ?first)
                 ((BloodKinFn parentOf) ?parent ?second)
                 (different ?first ?second))
            ((BloodKinFn siblingOf) ?first ?second)))))
(exceptWhen (gestationalOnlyParentOf ?parent ?second)
 (set/defaultRule
  (set/forwardRule
   (implies (and ((BloodKinFn parentOf) ?parent ?first)
                 ((BloodKinFn parentOf) ?parent ?second)
                 (different ?first ?second))
            ((BloodKinFn siblingOf) ?first ?second)))))
