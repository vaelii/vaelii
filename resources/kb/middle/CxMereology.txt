;; CxMereology — part/location/ownership theory: a part is located wherever
;; its whole is, and owning a whole entails owning its parts. Sees the shared
;; upper ontology via CxUniverse; CxWell sees it.

(genlCx CxMereology CxUniverse)
(genlCx CxWell CxMereology)

(set/forwardRule (implies (and (tangiblePartOf ?part ?whole) (locatedIn ?whole ?place))
                          (locatedIn ?part ?place)))

(set/forwardRule (implies (and (owns ?p ?whole) (tangiblePartOf ?part ?whole))
                          (owns ?p ?part)))

;; kb-integrity suggests a transitiveInArgInverse declaration for each rule.
(acknowledgedFinding :rule-macro (transitiveInArgInverse locatedIn 1 tangiblePartOf)
                     "An open locatedIn query reads the stored conclusions; the declaration answers no goal open at the part.")
(acknowledgedFinding :rule-macro (transitiveInArgInverse owns 2 tangiblePartOf)
                     "An open owns query reads the stored conclusions; the declaration answers no goal open at the part.")
