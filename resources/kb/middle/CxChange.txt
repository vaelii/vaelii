;; CxChange — the theory of change: a state persists until something ends it. A simple
;; event calculus over time.txt's vocabulary and the point algebra's ordering of instants.
;; Sees the shared upper ontology via CxUniverse; CxWell sees it.
;; The narrative vocabulary, the two directions and the stratification: docs/time.md,
;; "Change over time: events, fluents, and inertia".

(genlCx CxChange CxUniverse)
(genlCx CxWell CxChange)

;; ---- what ended a fluent, and when ---------------------------------------
;; Forward, so a stored order between named moments leaves the reason inertia stopped.
(set/forwardRule
 (implies (and (happens ?event ?instant)
               (terminates ?event ?fluent ?instant)
               (instantBefore ?instant1 ?instant)
               (instantBefore ?instant ?instant2))
          (clipped ?instant1 ?fluent ?instant2)))

;; The half-open reading, for a fluent that was the case at the outset: nothing named the
;; moment it began, so there is no lower bound to state.
(set/forwardRule
 (implies (and (happens ?event ?instant1)
               (terminates ?event ?fluent ?instant1)
               (instantBefore ?instant1 ?instant2))
          (clippedBefore ?fluent ?instant2)))

;; ---- inertia: a state persists until something ends it -------------------
;; Backward, and one literal of negation as failure over a terminating event not known to
;; lie outside the span, joined when the question is asked: the moment asked about may be
;; one no stored fact names.
(set/backwardRule
  (implies (and (happens ?event ?instant1)
                (initiates ?event ?fluent ?instant1)
                (instantBefore ?instant1 ?instant2)
                (unknown (thereExists (?ender ?instant)
                                      (and (happens ?ender ?instant)
                                           (terminates ?ender ?fluent ?instant)
                                           (unknown (equals ?instant ?instant1))
                                           (unknown (instantBefore ?instant ?instant1))
                                           (unknown (instantNotAfter ?instant ?instant1))
                                           (unknown (equals ?instant ?instant2))
                                           (unknown (instantBefore ?instant2 ?instant))
                                           (unknown (instantNotBefore ?instant ?instant2))))))
           (holdsAt ?fluent ?instant2)))

;; And the same for a fluent nobody started, because it was already the case. time_point
;; is the generator: negation as failure is closed, so the moment being asked about has to
;; be bound by something positive before the unknown can run (docs/naf.md).
(set/backwardRule
  (implies (and (initially ?fluent)
                (time_point ?instant2)
                (unknown (thereExists (?ender ?instant1)
                                      (and (happens ?ender ?instant1)
                                           (terminates ?ender ?fluent ?instant1)
                                           (unknown (equals ?instant1 ?instant2))
                                           (unknown (instantBefore ?instant2 ?instant1))
                                           (unknown (instantNotBefore ?instant1 ?instant2))))))
           (holdsAt ?fluent ?instant2)))

;; ---- when an input and an output exist, against their event ----------------
;; Each rule concludes a point relation over (StartFn X) and (EndFn X), which the point
;; network reads (docs/time.md, "When an input and an output exist").
(set/forwardRule
 (implies (and (output ?e ?x))
          (and (instantNotBefore (StartFn ?x) (StartFn ?e))
               (instantNotAfter (StartFn ?x) (EndFn ?e)))))
(set/forwardRule
 (implies (and (destroyedInput ?e ?x))
          (and (instantNotBefore (EndFn ?x) (StartFn ?e))
               (instantNotAfter (EndFn ?x) (EndFn ?e)))))
(set/forwardRule
 (implies (and (input ?e ?x))
          (instantNotAfter (StartFn ?x) (EndFn ?e))))
(set/forwardRule
 (implies (and (preservedInput ?e ?x))
          (instantAfter (EndFn ?x) (EndFn ?e))))
