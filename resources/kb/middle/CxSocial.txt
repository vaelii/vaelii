;; CxSocial — theory over the social relations: what acquaintance follows from,
;; how employment relates to membership, and the general relationship and
;; dwelling vocabulary over two persons. Sees the shared upper ontology via
;; CxUniverse; CxWell sees it. CxTra, an opt-in context below it, states the
;; relationship and plural-identity vocabulary narrower than this.
;;
;; Every rule and genl edge here runs one way only. `knows` is deliberately not
;; symmetric — a great many people know someone who does not know them — so friendship
;; and marriage, which *are* symmetric, each yield acquaintance in both directions
;; through their own symmetry rather than through a symmetry `knows` does not have. Stating (symmetric
;; knows) would have been one word and would have made the KB assert the converse of
;; every acquaintance it ever learned.

(genlCx CxSocial CxUniverse)
(genlCx CxWell CxSocial)

(genl friendOf knows)

(genl marriedTo knows)

;; Employment is one way of belonging to a body of people; belonging is not a way of
;; being employed by it, so there is no converse edge.
(genl worksFor memberOf)

;; ---- relationships: family, romance, work ---------------------------------
;; coworkerOf and roommateOf each specialize knows, the way friendOf and marriedTo
;; do above, and kinOf between persons reads knows by default; romanticPartnerOf specializes
;; friendOf. No edge
;; relates marriedTo to romanticPartnerOf: friendOf and marriedTo reach knows
;; by separate genl edges above because this theory does not claim every
;; marriage is a friendship, and a marriedTo-under-romanticPartnerOf-under-friendOf
;; chain would carry marriedTo under friendOf too.
;; Two relatives need not have met: a great-aunt who died before a child was born is
;; still the child's relative.  So knows follows from kinOf by default, and a
;; stated denial overrides it.  The rule-macro pass suggests the genl edge the rule
;; would be if it were strict, and the suggestion is declined.
(set/defaultRule (set/forwardRule (implies (and (kinOf ?x ?y) (person ?x) (person ?y)) (knows ?x ?y))))
(declined_rule_macro (Quote (genl kinOf knows)))
(implementationNote kinOf "Between persons, knows follows by a default rule, not a genl edge: two relatives need not have met, and non-person organisms need not know anyone. The rule-macro suggestion (genl kinOf knows) is recorded declined.")

(comment romanticPartnerOf
         "(romanticPartnerOf ?x ?y) means ?x and ?y are in a romantic relationship: dating, life partnership, or any other romantic connection.")
(set/monotonic (instance_relation_predicate romanticPartnerOf))
(set/monotonic (symmetric romanticPartnerOf))
(arg romanticPartnerOf 1 person)
(arg romanticPartnerOf 2 person)
(genl romanticPartnerOf friendOf)

(comment coworkerOf "(coworkerOf ?x ?y) means ?x and ?y work together.")
(set/monotonic (instance_relation_predicate coworkerOf))
(set/monotonic (symmetric coworkerOf))
(set/monotonic (time_varying_relation coworkerOf))
(arg coworkerOf 1 person)
(arg coworkerOf 2 person)
(genl coworkerOf knows)

(comment roommateOf
         "(roommateOf ?x ?y) means ?x and ?y live together. Does not imply romantic involvement.")
(set/monotonic (instance_relation_predicate roommateOf))
(set/monotonic (symmetric roommateOf))
(set/monotonic (time_varying_relation roommateOf))
(arg roommateOf 1 person)
(arg roommateOf 2 person)
(genl roommateOf knows)

(comment originatorOf
         "(originatorOf ?originator ?originated) means ?originator brought ?originated into being.")
(set/monotonic (instance_relation_predicate originatorOf))
(set/monotonic (eternal_predicate originatorOf))
(set/monotonic (asymmetric originatorOf))
(arg originatorOf 1 causal)
(arg originatorOf 2 causal)

;; The provenance of gestational-only parenthood is left open by this theory.
(exceptWhen (gestationalOnlyParentOf ?parent ?child)
  (set/backwardRule
    (implies ((BloodKinFn parentOf) ?parent ?child)
             (originatorOf ?parent ?child))))

(comment relationshipLabel
         "(relationshipLabel ?namer ?named ?label) means ?namer calls their relationship with ?named by ?label. Perspectival: the label ?named would give the same relationship need not be the same string.")
(ternary_predicate relationshipLabel)
(set/monotonic (time_varying_relation relationshipLabel))
(arg relationshipLabel 1 person)
(arg relationshipLabel 2 person)
(quotedArg relationshipLabel 3 string)

;; ---- dwellings --------------------------------------------------------------
;; A dwelling is a residential unit: a house, an apartment, a room, not a city or a
;; country. An apartment and a room are parts of a building and not buildings, so a
;; dwelling is a container, abstract.txt's, and container's disjoints carry it clear of
;; organism, substance and person. A detached house is a dwelling and a building at
;; once, and a motorhome a dwelling and a vehicle (the orthogonals below). dwellsIn is a
;; spec of livesIn restricted to a human and a dwelling specifically; two different
;; people who dwell in the same one are roommates by the rule below.

(unary_predicate dwelling)
(comment dwelling
         "A residential unit.")
(set/monotonic (time_varying_relation dwelling))
(positiveExampleComment dwelling "a house")
(positiveExampleComment dwelling "an apartment")
(positiveExampleComment dwelling "a room")
(negativeExampleComment dwelling "a city")
(negativeExampleComment dwelling "a country")
(set/monotonic (genl dwelling container))
(disjoint dwelling clothing)
(disjoint dwelling furniture)
;; A detached house is a dwelling and a building, and a motorhome a dwelling and a vehicle.
;; Demolition ends the house as both, and scrapping ends the motorhome as both.
(orthogonal dwelling building)
(orthogonal dwelling vehicle)

(comment dwellsIn
         "(dwellsIn ?person ?dwelling) means ?person lives in ?dwelling specifically. Derives roommateOf for two different people who dwell in the same one.")
(set/monotonic (instance_relation_predicate dwellsIn))
(arg dwellsIn 1 human)
(arg dwellsIn 2 dwelling)
(genl dwellsIn livesIn)

(set/forwardRule
  (implies (and (dwellsIn ?x ?d) (dwellsIn ?y ?d) (different ?x ?y))
           (roommateOf ?x ?y)))
