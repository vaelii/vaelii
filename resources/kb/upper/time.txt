;; time.txt — the theory of time: qualitative relations between INTERVALS in Allen's
;; algebra, qualitative order between INSTANTS in the point algebra, the two joined by an
;; interval's endpoints, and the metric layer that puts real measures on the gaps. Loaded
;; into CxUniverse with the rest of kb/upper/. An interval and an instant are both
;; ordinary individuals, and nothing here declares them.
;; The two algebras, the calendar terms and the event-calculus vocabulary: docs/time.md;
;; the metric layer: docs/stp.md and docs/duration.md.

(comment cxTimeOf
         "(cxTimeOf ?context ?time) means that ?time is the time ?context covers, the time (CxTimeOfFn ?context) names. CxTimeOfFn is the corresponding function, and its value is the last argument. See docs/time.md.")
(binary_predicate cxTimeOf)
(instance_relation_predicate cxTimeOf)
(arg cxTimeOf 1 context)
(arg cxTimeOf 2 time)

;; A slice of a slice is the slice of the inner slice's context at the intersection of
;; the two times:
;; (CxTimeFn (CxTimeFn ?context ?time1) ?time2) is the context
;; (CxTimeFn ?context (TimeIntersectionFn ?time1 ?time2)).  The two parts state it
;; through cxMonadOf and cxTimeOf.  Each rule reads the natFormula of the slice's cx/
;; constant, so a slice written with an inner slice's cx/ constant answers through them;
;; an inner slice spelled as a CxTimeFn application is a term, not a constant, and does
;; not.  A context that is not a slice is its own monad, and a slice has the monad of the
;; context it slices.
(set/backwardRule
 (implies (and (monad_context ?monad))
          (cxMonadOf ?monad ?monad)))
(set/backwardRule
 (implies (and (natFormula ?slice (CxTimeFn ?context ?time)) (cxMonadOf ?context ?monad))
          (cxMonadOf ?slice ?monad)))
;; A slice of a context that is not a slice covers its own time, and a slice of a slice
;; covers the intersection of the inner slice's time and its own.
(set/backwardRule
 (implies (and (natFormula ?slice (CxTimeFn ?monad ?time)) (monad_context ?monad))
          (cxTimeOf ?slice ?time)))
(set/backwardRule
 (implies (and (natFormula ?slice (CxTimeFn ?context ?time))
               (natFormula ?context (CxTimeFn ?inner ?innerTime))
               (cxTimeOf ?context ?outer))
          (cxTimeOf ?slice (TimeIntersectionFn ?outer ?time))))
;; The equality itself, between the two spellings.  It concludes about CxTimeFn
;; applications, so it answers no goal until the engine reads it (docs/nat.md, a rule or
;; a goal that names an open application).
(set/backwardRule
 (implies (and (context ?context) (time ?time1) (time ?time2))
          (equals (CxTimeFn (CxTimeFn ?context ?time1) ?time2)
                  (CxTimeFn ?context (TimeIntersectionFn ?time1 ?time2)))))

(comment Forever

;; Every interval falls within Forever, and every instant lies inside it.  Backward:
;; answered when a query asks, never stored for every interval in the KB.
(set/backwardRule
 (implies (and (time_interval ?interval))
          (subintervalOf ?interval Forever)))
;; kb-integrity pairs the root denial of a proper part's whole being its part
;; (abstract.txt) with that definition, as a rule clash.  Acknowledged rather than
;; excepted, for the reason the record gives.
(acknowledgedFinding :rule-clash
  (RulePairFn
   (Quote (implies (properPartOf ?part ?whole) (not (partOf ?whole ?part))))
   (Quote (implies (and (time_interval ?interval))
                    (subintervalOf ?interval Forever))))
  "Left unexcepted on purpose. Forever's definition concludes that every time interval is a subinterval of it, so a part, and the root rule denies one. Both fire for one tuple only where the facts already contradict the root rule: Forever stated a proper part of an interval. Both rules are strict definitions, so that clash is the one the facts owe, and an exception would hide it.")
(set/backwardRule
 (implies (and (time_point ?instant))
          (includesInstant Forever ?instant)))

;; A monad context covers all of its time.
(set/backwardRule
 (implies (and (monad_context ?monad))
          (cxTimeOf ?monad Forever)))
