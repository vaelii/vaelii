;; organism.txt — the biological upper-ontology types: animal/plant and their
;; kinds (mammal, bird, fish, reptile, amphibian, insect, arachnid, tree, flower, …)
;; down to the individual species used across the fables. Loaded into CxUniverse with the
;; rest of kb/upper/.
;;
;; The five vertebrate classes are separated by one `vertebrate_class` metatype rather
;; than by ten `disjoint` sentences. Nothing is stored for the pairs — the clique is
;; consulted — so retracting the metatype releases all ten at once, and retracting one
;; membership releases exactly that class's pairs and leaves the rest separated.
;;
;; animal and plant are placed under organism and separated from each other by one
;; (separating organism animal plant) declaration. The declaration states both genl edges and the
;; separation, so no (genl plant organism) or (disjoint animal plant) sentence is written
;; beside it. core.txt also stores (genl animal organism) as a sentence.
;;
;; The folk taxonomy sorts the organisms as everyday speech does: a species-level kind
;; (cat, oak) under a class-level kind (mammal, tree), and an animal class either with a
;; backbone or without one.  Each level is a disjoint_metatype, so its members are pairwise
;; disjoint without a sentence per pair, and every kind below a member inherits the
;; separation: folk_species separates every two species, so no organism is a cat and a
;; dog, and none is a spider and an owl.

(comment adultFormOf
         "(adultFormOf ?stage ?kind) means that ?stage is a kind of ?kind in its adult stage: (adultFormOf butterfly lepidopteran) says butterflies are adult lepidopterans. A claim about kinds, like partType: it says what the stage is a stage of and nothing about any particular organism, and (genl ?stage ?kind) is stated beside it, because the organism goes on existing through the change. Companion to larvalFormOf.")
(set/monotonic (type_relation_predicate adultFormOf))
(genlArg adultFormOf 1 organism)
(genlArg adultFormOf 2 organism)

;; animal.
(set/monotonic (partition animal vertebrate invertebrate))

(comment folk_biological_class
         "(folk_biological_class ?type) means that ?type is a class-level kind of organism as everyday speech names it, broader than a species: mammal, insect, tree. The parent of vertebrate_class, invertebrate_class and plant_class, which separating keeps apart, so no class is of two of them.")
(unary_predicate folk_biological_class)
(genlArg folk_biological_class 1 organism)
(metatype folk_biological_class)
(set/monotonic (time_invariant_relation folk_biological_class))
;; typeGenl requires its target to be a first-order type: (interArg typeGenl 1 metatype 2 type).
(type organism)
(typeGenl folk_biological_class organism)
;; Every member is a first-order type, whose instances are individual organisms.
(genl folk_biological_class type)
;; A mammal or a tree is one for as long as it exists, so every folk biological class is
;; time-invariant.
(set/monotonic (genl folk_biological_class time_invariant_relation))
;; The separating declaration installs each part's genl edge to folk_biological_class.
(set/monotonic (separating folk_biological_class vertebrate_class invertebrate_class plant_class))

(comment folk_species
         "(folk_species ?type) means that ?type is a species-level kind of organism as everyday speech names it. A disjoint_metatype, so no organism is of two species, and never a class-level kind: a species is not a folk_biological_class.")
(positiveExample folk_species (Quote (folk_species cat)))
(positiveExample folk_species (Quote (folk_species dog)))
(positiveExample folk_species (Quote (folk_species oak)))
(positiveExample folk_species (Quote (folk_species human)))
(genlArg folk_species 1 organism)
(metatype folk_species)
(typeGenl folk_species organism)
;; Every member is a first-order type, whose instances are individual organisms.
(genl folk_species type)
;; A mouse is a mouse for as long as it exists, so every folk species is time-invariant.
(set/monotonic (genl folk_species time_invariant_relation))
(set/monotonic (disjoint_metatype folk_species))
(set/monotonic (disjoint folk_biological_class folk_species))
;; A species membership is definitional, so each is written :monotonic and folk_species
;; is forced-monotonic: a membership is never the loser of a nogood, a denial
;; of one is inert, and a rule concluding a forced-monotonic literal such as orthogonal
;; from (folk_species ?s) alone is a forced-monotonic rule whose firings are believed
;; (docs/nmtms.md).
(forced_monotonic_predicate folk_species)
(set/monotonic (time_invariant_relation folk_species))
